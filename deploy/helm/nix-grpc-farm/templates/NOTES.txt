Client certificate from the farm CA (CN must match auth.accessRules):

  kubectl -n {{ .Release.Namespace }} apply -f - <<EOF
  apiVersion: cert-manager.io/v1
  kind: Certificate
  metadata: {name: ci}
  spec: {secretName: ci-tls, commonName: ci-1, usages: [client auth], issuerRef: {name: {{ $svc }}}}
  EOF
  for f in ca.crt tls.crt tls.key; do kubectl -n {{ .Release.Namespace }} get secret ci-tls -o go-template="{{ "{{" }}index .data \"$f\" | base64decode}}" > $f; done
{{- end }}
